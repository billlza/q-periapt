def require(value: bool, message: str) -> None:
    if not value:
        raise ValueError(message)


def canonical_sha(value: object) -> str:
    raw = json.dumps(value, sort_keys=True, separators=(',', ':')).encode()
    return hashlib.sha256(raw).hexdigest()


def identity(path: Path) -> dict:
    require(path.is_file() and not path.is_symlink(), 'not a regular file: ' + str(path))
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        while block := stream.read(1024 * 1024):
            digest.update(block)
    return {'bytes': path.stat().st_size, 'sha256': digest.hexdigest()}


def inventory(root: Path) -> dict:
    entries = {}
    for path in sorted(root.rglob('*')):
        require(not path.is_symlink(), 'unexpected symbolic link: ' + str(path))
        if path.is_dir():
            continue
        entries[path.relative_to(root).as_posix()] = identity(path)
    return entries


def write_json(path: Path, value: object) -> None:
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, sort_keys=True)
        stream.write('\n')


def strict_json(data: bytes) -> object:
    def unique(pairs):
        value = {}
        for key, item in pairs:
            require(key not in value, 'duplicate JSON field: ' + key)
            value[key] = item
        return value
    def reject_constant(value):
        raise ValueError('nonfinite JSON value: ' + value)
    return json.loads(data, object_pairs_hook=unique, parse_constant=reject_constant)

def require_exclusive_resources() -> dict:
    memory = Path('/proc/meminfo').read_text()
    match = re.search(r'^MemAvailable:\s+([0-9]+) kB$', memory, re.MULTILINE)
    require(match is not None, 'cannot read current MemAvailable')
    available = int(match[1]) * 1024
    require(available >= 7 * 1024**3, 'less than the declared 7 GiB available-memory admission')
    active = []
    for entry in Path('/proc').iterdir():
        if not entry.name.isdigit():
            continue
        try:
            command = (entry / 'cmdline').read_bytes()
        except (FileNotFoundError, ProcessLookupError):
            continue
        except PermissionError:
            continue
        if b'com.semmle.cli2.CodeQL' in command:
            active.append(int(entry.name))
    require(not active, 'another CodeQL JVM is active: ' + str(active))
    return {'observed_utc': datetime.now(timezone.utc).isoformat(),
            'mem_available_bytes': available, 'active_codeql_jvms': active}

    require(not output.exists() and not output.is_symlink(), 'output directory already exists')

    require(not (database / 'results').exists(), 'private database has previous results')
    database_before = production_plan.admit_database(database)

    class JvmCommandSampler(ProcessResourceSampler):
        """Retain actual JVM argv from the existing owned process-tree sample."""
        def __init__(self, directory):
            super().__init__(directory)
            self.jvm_commands = {}
            self.jvm_read_errors = []
            self.jvm_lock = threading.Lock()

        def _sample(self):
            sample = super()._sample()
            if sample is None:
                return sample
            for process in sample['processes']:
                if process['comm'] != 'java':
                    continue
                try:
                    raw = read_small(self.proc_root / str(process['pid']) / 'cmdline')
                    current = self._stat(process['pid'])
                    require(current['start_ticks'] == process['start_ticks'], 'JVM process identity changed')
                    argv = [part.decode('utf-8', 'surrogateescape') for part in raw.split(b'\0') if part]
                    if 'com.semmle.cli2.CodeQL' not in argv:
                        continue
                    entry = argv.index('com.semmle.cli2.CodeQL')
                    key = (process['pid'], process['start_ticks'], hashlib.sha256(raw).hexdigest())
                    with self.jvm_lock:
                        require(key in self.jvm_commands or len(self.jvm_commands) < 64,
                                'JVM command observation count exceeded')
                        if key not in self.jvm_commands:
                            self.jvm_commands[key] = {
                                'pid': process['pid'], 'start_ticks': process['start_ticks'],
                                'parent_pid': process['ppid'], 'root_pid': process['root_pid'],
                                'root_start_ticks': process['root_start_ticks'],
                                'first_observed_elapsed_seconds': sample['elapsed_seconds'],
                                'cmdline_sha256': key[2], 'argv': argv,
                                'jvm_heap_arguments': [arg for arg in argv[:entry] if arg.startswith('-Xmx')],
                                'cli_arguments': argv[entry + 1:]}
                except (OSError, ValueError) as error:
                    with self.jvm_lock:
                        if len(self.jvm_read_errors) < 32:
                            self.jvm_read_errors.append({'pid': process['pid'], 'type': type(error).__name__,
                                                         'message': str(error)})
            return sample

        def command_receipt(self):
            with self.jvm_lock:
                return {'schema': 1, 'method': 'Actual argv from sampled owned process-tree identities',
                        'commands': list(self.jvm_commands.values()), 'read_errors': list(self.jvm_read_errors),
                        'sampling_state': dict(self.state),
                        'scope': 'All observed wrapper and evaluator JVM commands, without deduplication of distinct process identities or argv'}

    try:
        resolved = strict_json(run.command('resolve-local-ram', [
            codeql, 'resolve', 'ram', '--dataset=' + str(database / 'db-rust'),
            *MEMORY_ARGUMENTS, '-J-Xmx512M', '--format=json'], seconds=120))
        write_json(evidence / 'LOCAL_MEMORY.json', {'resolved': resolved, 'expected': RESOLVED_MEMORY,
                   'scope': 'Local explicit-Xmx method; not the hosted allocation'})
        require(resolved == RESOLVED_MEMORY, 'actual local memory resolution differs; no query was started')
        require_exclusive_resources()
        try:
            sampler = JvmCommandSampler(evidence)
            sampler.start()
            resource_observation = {'status': 'started'}
        except Exception as error:
            resource_observation = {'status': 'unavailable', 'type': type(error).__name__,
                                    'message': str(error)}
        run.command('analyze', analysis_command, seconds=MAXIMUM_SECONDS,
                    timeout_policy=TimeoutPolicy.STANDARD)
    except BaseException as error:
        failure = error
    finally:
        if sampler is not None:
            try:
                sampler.stop()
                if resource_observation['status'] == 'started':
                    resource_observation = {'status': 'completed'}
            except BaseException as error:
                resource_observation = {'status': 'stop_failed', 'type': type(error).__name__,
                                        'message': str(error)}
        write_json(evidence / 'RESOURCE_OBSERVATION_STATUS.json', resource_observation)
        if sampler is not None:
            write_json(evidence / 'JVM_COMMANDS.json', sampler.command_receipt())
        for label, check in (
            ('plan-postcheck', lambda: require(projection.validate(pack) == plan_before,
                                              'projection identity changed')),
            ('pack-postcheck', lambda: require(inventory(pack) == pack_before, 'query pack changed')),
            ('runtime-postcheck', lambda: require(production_plan.runtime_inventory(codeql_root) == runtime_before,
                                                 'analysis runtime changed')),
            ('qlconfig-postcheck', lambda: require(identity(qlconfig) == qlconfig_before,
                                                  'private QL configuration changed')),
            ('source-database-postcheck', lambda: require(inventory(DATABASE) == initial_database,
                                                        'accepted source database changed')),
        ):
            try:
                check()
            except BaseException as error:
                post_errors.append({'step': label, 'type': type(error).__name__, 'message': str(error)})
        write_json(evidence / 'POSTCHECK.json', {'errors': post_errors,
                   'original_failure': None if failure is None else {'type': type(failure).__name__, 'message': str(failure)}})

    if failure is not None:
        for error in post_errors:
            failure.add_note(str(error))
        raise failure
    require(not post_errors, 'post-execution identity check failed')
    observations = evaluator_logs.observe(evidence, pattern='execute-queries-*.log', stdio_prefix='analyze',
                                          capture_invocation=True)
    write_json(evidence / 'EVALUATOR_OBSERVATIONS.json', observations)
    evaluator_logs.validate_loads(observations, [str((pack / name).with_suffix('.qlx')) for name in queries])
    require(observations['evaluator_start_lines'] == 1
            and observations['off_heap_mb_observed'] == [NATIVE_MIB], 'actual evaluator allocation differs')
    require(observations['query_starts'] == observations['query_completions'] == 39,
            'evaluator progress does not show the complete 39-query execution')

    require(heap_arguments and all(value == HEAP_MIB for value in heap_arguments),
            'actual evaluator JVM heap arguments differ from the admitted local allocation')
    heap_observations = observations['max_heap_bytes_observed']
    require(heap_observations and set(heap_observations) == {HEAP_MIB * 1024**2},
            'actual JVM heap observations differ from the admitted local allocation')

    result_root = database / 'results/codeql/rust-queries'
    expected_results = {str(Path(name).with_suffix('.bqrs')) for name in queries}
    found_results = {p.relative_to(result_root).as_posix() for p in result_root.rglob('*.bqrs')}
    require(found_results == expected_results and len(list(database.rglob('*.bqrs'))) == 39,
            'fresh BQRS census differs from the exact 39-query suite')
    results = {name: identity(result_root / name) for name in sorted(expected_results)}
    write_json(evidence / 'FRESH_RESULTS.json', results)
    for name in sorted(results):
        target = evidence / 'results' / name
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(result_root / name, target)
    sarif = strict_json((evidence / 'results.sarif').read_bytes())
    profile.validate_sarif(sarif, 'native')

    decoded = run.command('decode-summary', [codeql, 'bqrs', 'decode', '--format=json',
                            '-J-Xmx512M', str(result_root / Path(SUMMARY).with_suffix('.bqrs'))], seconds=120)
    decoded_value = strict_json(decoded)
    require(all(decoded_value == value for value in baseline_values.values()),
            'complete SummaryStats result differs from a retained baseline')
    require(canonical_sha(decoded_value) == BASELINE_SEMANTIC_SHA256, 'SummaryStats semantic identity differs')
    rows = decoded_value['#select']['tuples']
    require(len(rows) == 29 and len({row[0] for row in rows}) == 29, 'SummaryStats row census differs')
    values = dict(rows)
    require(values['Taint sources - active'] == 15 and values['Taint reach - nodes tainted'] == 106,
            'accepted fixture no longer has the established nonempty taint population')

    database_after = production_plan.admit_database(database)
    metadata_after = (database / 'codeql-database.yml').read_bytes()
    (evidence / 'DATABASE_METADATA_AFTER.yml').write_bytes(metadata_after)
    write_json(evidence / 'DATABASE_ADMISSION_AFTER.json', database_after)
    before_fields, after_fields = metadata_root_fields(metadata_before.decode()), metadata_root_fields(metadata_after.decode())
    before_fields.pop('baselineLinesOfCode', None)
    after_fields.pop('baselineLinesOfCode', None)
    require(before_fields == after_fields, 'database structural metadata changed')
    immutable = {name: expected for name, expected in initial_database.items()
                 if not name.startswith(('db-rust/default/cache/', 'log/', 'working/', 'temp/'))
                 and name != 'codeql-database.yml'}
    require(all(identity(database / name) == expected for name, expected in immutable.items()),
            'immutable extracted input bytes changed')
    require(projection.validate(pack) == plan_before and inventory(pack) == pack_before,
            'query pack changed during result decoding')
    require(production_plan.runtime_inventory(codeql_root) == runtime_before,
            'runtime changed during result decoding')
    require(identity(qlconfig) == qlconfig_before, 'private QL configuration changed during decoding')
    require(not subprocess.check_output(['git', 'status', '--porcelain'], cwd=repo).strip()
            and subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=repo, text=True).strip() == actual_commit,
            'method source checkout changed')
    write_json(evidence / 'LOCAL_WRAPPER_RESULT.json', {
        'schema': 1, 'method_version': METHOD_VERSION,
        'kind': 'qperiapt.local_nonempty_fixture_database_analyze',
        'status': 'complete_equal_summary_valid_sarif', 'source_commit': actual_commit,
        'finished_utc': datetime.now(timezone.utc).isoformat(),
        'projection_identity': plan_before, 'query_count': 39,
        'precompiled_query_loads': 39, 'implicit_compilations': 0, 'fresh_bqrs_count': 39,
        'sarif': identity(evidence / 'results.sarif'),
        'sarif_validation': 'existing codeql_rust_profile.validate_sarif with native category',
        'summary_bqrs': results[str(Path(SUMMARY).with_suffix('.bqrs'))],
        'summary_decoded': identity(evidence / 'decode-summary.stdout'),
        'summary_semantic_sha256': canonical_sha(decoded_value), 'summary_rows': 29,
        'all_retained_baselines_equal': True, 'retained_baselines_rerun': False,
        'summary_values': values, 'immutable_initial_database_files': len(immutable),
        'plan_pack_and_runtime_unchanged_after_analysis_and_decode': True,

def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--execute', action='store_true', help='Run only after the exclusive large probe has ended')
    parser.add_argument('--repository', type=Path, default=WORKSPACE / 'summary-production-implementation')
    parser.add_argument('--source-commit')
    parser.add_argument('--output', type=Path, default=WORKSPACE / 'summary-stage34-analyze-fixture')
    args = parser.parse_args()
    if not args.execute:
        print(json.dumps(declared_method(), indent=2))
        return 0
    require(args.source_commit is not None, '--source-commit is required for execution')
    try:
        return execute(args)
    except BaseException as error:
        evidence = args.output.resolve() / 'evidence'
        if evidence.is_dir() and not (evidence / 'FAILURE.json').exists():
            write_json(evidence / 'FAILURE.json', {'type': type(error).__name__, 'message': str(error),
                       'notes': getattr(error, '__notes__', []),
                       'recorded_utc': datetime.now(timezone.utc).isoformat()})
        print(type(error).__name__ + ': ' + str(error), file=sys.stderr)
        command_receipt = evidence / 'analyze.json'
        if command_receipt.is_file():
            record = strict_json(command_receipt.read_bytes())
            code = record.get('returncode')
            if isinstance(code, int) and 1 <= code <= 255:
                return code
        return 1

    raise SystemExit(main())
