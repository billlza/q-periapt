from pathlib import Path
import ast
import datetime
import hashlib
import json
import subprocess
import sys

assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
ROOT = Path(__file__).resolve().parents[2]
OUT = Path(__file__).resolve().parent
path = ROOT / 'artifact/continuity_kotlin_consumer.py'
old = subprocess.run(['git', 'show', 'HEAD:artifact/continuity_kotlin_consumer.py'], cwd=ROOT,
    stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=True).stdout
new = path.read_bytes()
before, after = ast.parse(old), ast.parse(new)

def inventory(tree):
    found = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Dict):
            for index, key in enumerate(node.keys):
                if isinstance(key, ast.Constant) and key.value == 'negative_controls':
                    expression = node.values[index]
                    assert isinstance(expression, ast.BinOp) and isinstance(expression.op, ast.Add)
                    assert ast.unparse(expression.left) == 'sorted(negatives)'
                    labels = ast.literal_eval(expression.right)
                    found.append(labels)
                    node.values[index] = ast.Constant(value='verified inventory-only change')
    assert len(found) == 1 and len(found[0]) == len(set(found[0]))
    return found[0]

old_labels, new_labels = inventory(before), inventory(after)
assert ast.dump(before, include_attributes=False) == ast.dump(after, include_attributes=False)
mapping = {
    'raw-owner': 'raw-owner-construction', 'raw-device': 'raw-device-construction',
    'raw-setup': 'raw-setup-construction', 'raw-enrollment': 'raw-enrollment-construction',
    'raw-native-owner': 'raw-native-owner-construction',
    'raw-retired': 'raw-retired-enrollment-construction',
    'RawAccountAuthorityProbe': 'raw-account-authority-construction',
    'RawConfigurationProbe': 'raw-configuration-construction',
    'RawConfigurationKeyProbe': 'raw-configuration-key-access',
}
assert set(new_labels) == set(mapping.values()) and new_labels[:len(old_labels)] == old_labels
data_root = ROOT / 'target/continuity-profile-coverage-20261105'
retention = json.loads((data_root / 'KOTLIN_RETENTION.json').read_bytes())
observations = {}
for profile in ('debug', 'release'):
    report_path = data_root / ('kotlin-' + profile) / 'identity-01/expanded-selected/KOTLIN_CONSUMER.json'
    data = report_path.read_bytes()
    row = json.loads(data)['profiles'][profile]
    controls = retention['profiles'][profile]['negative_controls']
    loader = set(controls) - set(mapping)
    assert loader == {'missing-property', 'relative-path', 'missing-library', 'directory-library',
                      'denied-native-access', 'missing-symbol'}
    assert row['negative_controls'] == sorted(loader) + old_labels
    actual_labels = {mapping.get(name, name) for name in controls}
    assert actual_labels == loader | set(new_labels)
    omitted = sorted(actual_labels - set(row['negative_controls']))
    assert len(omitted) == 4
    observations[profile] = {'original_report_sha256': hashlib.sha256(data).hexdigest(),
        'original_summary_labels': len(row['negative_controls']), 'original_refusal_receipts': len(controls),
        'omitted_original_labels': omitted, 'corrected_inventory_matches_all_receipts': True}
result = {'verified_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'original_hosted_source': retention['source_commit'],
    'source_before_sha256': hashlib.sha256(old).hexdigest(), 'source_after_sha256': hashlib.sha256(new).hexdigest(),
    'ast_equal_after_removing_inventory': True, 'profiles': observations,
    'scope': 'Only report inventory adds four already-executed refusals. Original runtime outcomes and reports are unchanged. This is not a new full installed-package execution.',
    'release_claim_eligible': False}
with (OUT / 'READBACK.json').open('x') as stream:
    stream.write(json.dumps(result, indent=2, sort_keys=True) + '\n')
print(json.dumps(result))
