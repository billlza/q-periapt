def digest(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def save(name, value):
    (OUT / name).write_text(json.dumps(value, indent=2) + '\n')

def snapshot(directory):
    return {str(p.relative_to(directory)): digest(p) for p in sorted(directory.rglob('*'))
            if p.is_file() and not p.is_relative_to(directory / 'build')}

source = REPO / 'target/tls-client-hints-current/package-source'

def run(name, args, timeout=600, expect=0):
    started = time.monotonic()
    with (OUT / (name + '.stdout')).open('wb') as stdout, (OUT / (name + '.stderr')).open('wb') as stderr:
        result = subprocess.run(args, cwd=outside, env=env, stdout=stdout, stderr=stderr,
                                timeout=timeout, check=False)
    save(name + '.json', {'args': args, 'returncode': result.returncode,
                         'seconds': time.monotonic()-started})
    if (expect == 0 and result.returncode != 0) or (expect != 0 and result.returncode == 0):
        raise RuntimeError(f'{name}: unexpected exit {result.returncode}; see retained stderr')
    print(name + ': completed', flush=True)

    binaries = {}

    for label, features in [('baseline', []), ('tables', ['--features', 'precomputed'])]:

    blocks = []
    # Builds and lint complete before any timing; alternating fresh process order.
    for block in range(5):
        order = ['baseline', 'tables'] if block % 2 == 0 else ['tables', 'baseline']
        values = {}
        for label in order:

            lines = [json.loads(line) for line in (OUT/(name+'.stdout')).read_text().splitlines()]

            assert lines[0] == {'format': 1, 'precomputed_tables': label == 'tables', 'samples': 1000,

            for cell in lines[1:-1]:

                assert all(type(ns) is int and 0 < ns < 120_000_000_000 for ns in cell['ns'])
            values[label] = lines[1:-1]
        blocks.append({'block': block, 'order': order, 'values': values})

    rows = []
    for index, (op, size) in enumerate(expected_cells):
        row = {'operation': op, 'context_bytes': size}
        for percentile in (50, 95, 99):

            ratios = [b/a for a, b in zip(per_label['baseline'], per_label['tables'], strict=True)]

                                    'ratio_median': statistics.median(ratios), 'ratio_range': [min(ratios), max(ratios)],
