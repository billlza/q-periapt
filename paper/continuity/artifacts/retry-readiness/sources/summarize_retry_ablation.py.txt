#!/usr/bin/env python3
"""Generate exact paired tables from the independently recounted native traces."""
from __future__ import annotations

import argparse
import csv
import hashlib
import json
from pathlib import Path


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def change(before, after):
    return 100 * (after / before - 1)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("audit", type=Path)
    parser.add_argument("output", type=Path)
    args = parser.parse_args()
    audit = json.loads(args.audit.read_text())
    if audit["status"] != "passed" or audit["traces"] != 28:
        raise ValueError("completed independent audit required")
    comparisons = audit["comparisons"]
    if len(comparisons) != 14 or any(x["fresh_baseline_differences_from_retained"] for x in comparisons):
        raise ValueError("14 freshly reproduced historical baseline rows required")
    details = {(x["policy"], x["profile"], x["scenario"]): x for x in audit["details"]}
    output = args.output
    output.mkdir(parents=True, exist_ok=True)
    results = []
    for pair in comparisons:
        b, a = pair["baseline"], pair["sender_turns_4"]
        if any(x["opportunities"] != 2048 or x["offered"] != 2048 for x in (b, a)):
            raise ValueError("opportunity denominator differs")
        if any(x["derived_owner_A"] != x["derived_owner_B"] for x in (b, a)):
            raise ValueError("capture-owner counts differ; table needs separate columns")
        row = {"budget": int(pair["profile"].split("-")[-1]), "scenario": pair["scenario"]}
        for prefix, metrics, policy in (("immediate", b, "immediate"), ("age4", a, "sender-turns-4")):
            detail = details[policy, pair["profile"], pair["scenario"]]
            for metric in ("committed", "delivered", "uncommitted", "post_capture_committed", "derived_owner_A",
                           "emitted_bytes", "application_frames", "ack_frames", "control_frames",
                           "retry_application_frames", "all_exact_frame_retransmissions",
                           "durable_image_revision_increments", "physical_sync_data"):
                row[prefix + "_" + metric] = metrics[metric]
            row[prefix + "_committed_undelivered"] = metrics["committed"] - metrics["delivered"]
            row[prefix + "_derived_fraction"] = metrics["derived_owner_A"] / metrics["post_capture_committed"]
            row[prefix + "_delivered_per_opportunity"] = metrics["delivered"] / metrics["opportunities"]
            row[prefix + "_bytes_per_delivered"] = metrics["emitted_bytes"] / metrics["delivered"]
            row[prefix + "_sync_data_per_delivered"] = metrics["physical_sync_data"] / metrics["delivered"]
            row[prefix + "_retry_emissions_after_delivery"] = detail["retry_emissions_after_prior_logical_delivery"]
            row[prefix + "_retry_bytes_after_delivery"] = detail["retry_bytes_after_prior_logical_delivery"]
            row[prefix + "_confirmed_epochs"] = detail["final_confirmed_epochs"]
            row[prefix + "_sending_epochs"] = detail["final_sending_epochs"]
        for metric in ("committed", "delivered", "emitted_bytes", "retry_application_frames",
                       "durable_image_revision_increments", "physical_sync_data"):
            row[metric + "_delta"] = a[metric] - b[metric]
            row[metric + "_percent_change"] = change(b[metric], a[metric])
        row["bytes_per_delivered_percent_change"] = change(row["immediate_bytes_per_delivered"], row["age4_bytes_per_delivered"])
        row["sync_data_per_delivered_percent_change"] = change(row["immediate_sync_data_per_delivered"], row["age4_sync_data_per_delivered"])
        row["derived_count_delta"] = a["derived_owner_A"] - b["derived_owner_A"]
        row["derived_fraction_percentage_point_change"] = 100 * (row["age4_derived_fraction"] - row["immediate_derived_fraction"])
        results.append(row)
    bidirectional = [r for r in results if r["scenario"] != "one_way"]
    summary = {
        "schema": 1,
        "status": "passed",
        "audit_sha256": digest(args.audit),
        "script_sha256": digest(Path(__file__)),
        "paired_cases": len(results),
        "fresh_baseline_rows_equal_to_retained": 14,
        "non_one_way_delivered_increases": sum(r["delivered_delta"] > 0 for r in bidirectional),
        "non_one_way_delivered_percent_change_range": [min(r["delivered_percent_change"] for r in bidirectional), max(r["delivered_percent_change"] for r in bidirectional)],
        "non_one_way_bytes_per_delivered_decreases": sum(r["bytes_per_delivered_percent_change"] < 0 for r in bidirectional),
        "non_one_way_sync_data_per_delivered_percent_change_range": [min(r["sync_data_per_delivered_percent_change"] for r in bidirectional), max(r["sync_data_per_delivered_percent_change"] for r in bidirectional)],
        "rows": results,
        "interpretation": "Fixed 2048-opportunity service yields; total bytes and total durable operations are retained. Per-delivery ratios divide those totals by actual logical deliveries, including residual work. The combined experiment duration is not a per-service timing benchmark.",
    }
    (output / "comparison-analysis.json").write_text(json.dumps(summary, indent=2, sort_keys=True) + "\n")
    csv_rows = [{k: json.dumps(v, separators=(",", ":")) if isinstance(v, list) else v for k, v in r.items()} for r in results]
    with (output / "comparison-analysis.csv").open("w", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(csv_rows[0]))
        writer.writeheader()
        writer.writerows(csv_rows)
    lines = [
        "# Native retry-readiness: independent paired results", "",
        "The four-sender-opportunity readiness rule increases actual logical delivery in all 12 cases with bidirectional exchange or resumed exchange. The two one-way cases retain their original budget-limited service. All 14 newly executed immediate-policy rows reproduce the retained matched-comparison metrics exactly.", "",
        "Each case offers 2,048 32-byte requests in 2,048 directed transmission opportunities, with at most one 8,192-byte frame per opportunity. A data retry becomes ready when the sender's opportunity count is at least four greater than at its latest data transmission. That sender-local count advances at the beginning of selection, after the request/opportunity is recorded and before control planning. Every data emission resets its own retry age. Only retry readiness changes: the native protocol, captured-state predictor, budget, persistence, request/transport schedule, first-send operations, and control/ACK rules remain the same. Actual selection and the resulting control/ACK counts respond to the changed readiness.", "",
        "Both capture owners give the same derived count in every row. The numerator is the specified captured-current-epoch predictor's successful derivations; the denominator includes every post-capture first commitment. Capture occurs after opportunity 0's local/network actions and checkpoint, before opportunity 1 local work. Later-epoch eligibility is interpreted through the manuscript's computational theorem.", "",
        "## Service and captured-state derivation", "",
        "C/D is first commitments / actual logical deliveries. Each arrow compares immediate to age 4 under the same budget and schedule.", "",
        "| Budget | Schedule | C/D, immediate | C/D, age 4 | Delivered change | Derived / post-capture, immediate | Derived / post-capture, age 4 |", 
        "|---:|---|---:|---:|---:|---:|---:|",
    ]
    for r in results:
        lines.append(f"| {r['budget']} | {r['scenario']} | {r['immediate_committed']}/{r['immediate_delivered']} | {r['age4_committed']}/{r['age4_delivered']} | {r['delivered_percent_change']:+.2f}% | {r['immediate_derived_owner_A']}/{r['immediate_post_capture_committed']} | {r['age4_derived_owner_A']}/{r['age4_post_capture_committed']} |")
    lines += ["", "The age-4 lossy cases each end with one committed message whose first emission at opportunity 2044 is lost. The age-4 budget-64 asymmetric case ends with one committed message whose first emission at opportunity 2029 is suspended. These remain in the commitment and post-capture denominators, and are excluded from actual delivery. No tail repair or extra transmission opportunity is introduced.", "",
              "## Communication and persistence", "",
              "Bytes are all emitted wire bytes, including application, retry, ACK and control frames. Per-delivery costs divide full emitted-byte or physical-sync totals by actual deliveries. They therefore include the cost of residual committed work at the observation horizon.", "",
              "| Budget | Schedule | Emitted bytes, immediate → age 4 | Bytes / delivered, immediate → age 4 | Bytes / delivered change | Revisions, immediate → age 4 | Physical sync_data, immediate → age 4 |",
              "|---:|---|---:|---:|---:|---:|---:|"]
    for r in results:
        lines.append(f"| {r['budget']} | {r['scenario']} | {r['immediate_emitted_bytes']:,} → {r['age4_emitted_bytes']:,} | {r['immediate_bytes_per_delivered']:.2f} → {r['age4_bytes_per_delivered']:.2f} | {r['bytes_per_delivered_percent_change']:+.2f}% | {r['immediate_durable_image_revision_increments']:,} → {r['age4_durable_image_revision_increments']:,} | {r['immediate_physical_sync_data']:,} → {r['age4_physical_sync_data']:,} |")
    lines += ["", "## Application retry outcomes", "",
              "Post-delivery retry is an observer classification: the sender still awaits an accepted cumulative ACK. The age rule uses only that sender's opportunity count and last emission, without reading receiver delivery state.", "",
              "| Budget | Schedule | All application retries, immediate → age 4 | Retries emitted after prior logical delivery, immediate → age 4 | ACK frames, immediate → age 4 | Control frames, immediate → age 4 |",
              "|---:|---|---:|---:|---:|---:|"]
    for r in results:
        lines.append(f"| {r['budget']} | {r['scenario']} | {r['immediate_retry_application_frames']} → {r['age4_retry_application_frames']} | {r['immediate_retry_emissions_after_delivery']} → {r['age4_retry_emissions_after_delivery']} | {r['immediate_ack_frames']} → {r['age4_ack_frames']} | {r['immediate_control_frames']} → {r['age4_control_frames']} |")
    lines += ["", "## Research conclusion", "",
              "The intervention establishes that a substantial part of the immediate-policy service cost comes from retry scheduling: a sender-local readiness rule recovers service while preserving the native protocol and durability path. In alternating exchange, delivered messages increase from 688 to 971 at budget 32 and from 718 to 999 at budget 64; application retries fall from 604/579 to 16/8, all of whose receptions first deliver their message. The 592/572 application retries emitted after prior logical delivery disappear. Captured-current-epoch derivations remain 63/127, while the corresponding post-capture denominators grow from 687/717 to 970/998.", "",
              "The cost profile is schedule-dependent. Bytes per delivered message decrease in 10 of the 12 exchange cases, including 11.41%/19.70% in alternating exchange. At budget 32, lossy and reordered schedules increase bytes per delivered message by 14.37% and 3.94% while increasing actual delivery by 36.09% and 16.84%. Total persistence rises with additional service; alternating physical sync_data counts change from 14,219 to 20,015 and from 14,579 to 20,255, while counts per delivered message remain 20.67→20.61 and 20.31→20.28.", "",
              "One-way service remains 32/64 messages in epoch 0. Each run substitutes one 4,746-byte control emission for one 141-byte application retry, increasing total bytes by 4,605 without adding service. The offline-then-exchange captured-epoch count increases by one (32→33 and 64→65), while its fraction of post-capture commitments decreases because actual service grows. All captured-current-epoch counts continue to obey the same local remaining-allocation bound 2b−1.", "",
              "Retain the historical immediate policy as the matched comparison and add age 4 as the controlled native scheduling result. The demonstrated improvement is per opportunity under the stated deterministic schedules; it supports a concrete scheduling contribution with both service and cost reported. It does not choose a universal wall-clock timeout. An adaptive local readiness policy is a separate follow-up, with the present result already complete.", "",
              "## Reproduction", "",
              "Run audit_retry_ablation.py with the extracted continuity-retry-age-execution directory, retained historical results-49.json, and a new output JSON. Keep audit_matched_traces.py beside it. Then run this summarizer with that independent audit JSON and an output directory. The source-bound runner's build/execution logs and digest receipts bind the actual native execution; this public-trace audit recounts the retained observations.", "",
    ]
    (output / "RESULTS.md").write_text("\n".join(lines))
    print(json.dumps({k: v for k, v in summary.items() if k not in ("rows", "interpretation")}, sort_keys=True))


if __name__ == "__main__":
    main()
