#!/usr/bin/env python3
"""Independently recount both native policies from a completed public artifact."""
from __future__ import annotations

import argparse
import importlib.util
import json
from pathlib import Path

import audit_matched_traces as recount


BASE = "358b921ab3469ce49f166633fc9a762ff6ed7e3d"
MINIMUM = 4


def verify_ages(trace, minimum):
    turns = [0, 0]
    last = {}
    histogram = {}
    for event in trace:
        if event["event"] == "opportunity":
            turns[event["owner"]] += 1
        elif event["event"] == "emit" and event["kind"] in ("application", "retry"):
            key = event["owner"], event["logical"]
            if event["kind"] == "retry":
                recount.require(key in last, "retry has no prior emitted instance")
                age = turns[event["owner"]] - last[key]
                recount.require(age >= minimum, "independent retry readiness failure")
                histogram[age] = histogram.get(age,0)+1
            else:
                recount.require(key not in last, "duplicate first transmission")
            last[key] = turns[event["owner"]]
    return histogram


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("execution",type=Path)
    parser.add_argument("historical_results",type=Path)
    parser.add_argument("output",type=Path)
    args = parser.parse_args()
    root = args.execution.resolve()
    evidence = root/"evidence"
    prepared = recount.read(root/"preparation.json")
    recount.require(prepared["base_commit"] == BASE and prepared["minimum_sender_turns"] == MINIMUM,
                    "source or policy identity differs")
    before = {r["path"]:r for r in prepared["before"]}
    after = {r["path"]:r for r in prepared["after"]}
    recount.require(before.keys() == after.keys(), "source inventory paths changed")
    changed = [p for p in before if before[p] != after[p]]
    recount.require(changed == ["research/continuity-identity-candidate/src/durable/messages/tests.rs"],
                    "unexpected source derivative")
    drivers = recount.read(root/"drivers.json")
    recount.require(drivers["baseline_sha256"] == "8ca7ea6b0d17923d16eff6f48c2a1ac07ce2eb5929c4b4526510e41a50db0fca"
                    and drivers["retry_age_sha256"] == recount.digest(root/"retry_age_driver.rs.txt")
                    and drivers["minimum_sender_turns"] == MINIMUM, "driver identity differs")
    executable = recount.read(root/"executable.json")
    recount.require(set(executable["exact_tests"]) == {"immediate","sender-turns-4"},
                    "exact native test inventory differs")
    manifest = recount.read(evidence/"manifest.json")
    recount.require(manifest["status"] == "complete" and len(manifest["files"]) == 29,
                    "incomplete ablation manifest")
    for item in manifest["files"]:
        relative = Path(item["path"])
        recount.require(not relative.is_absolute() and ".." not in relative.parts, "manifest path escape")
        path = evidence/relative
        recount.require(path.stat().st_size == item["bytes"] and recount.digest(path) == item["sha256"],
                        "public ablation evidence changed")
    report = recount.read(evidence/"retry-age-report.json")
    recount.require(report["status"] == "passed" and report["execution"] == "native_compiled_and_executed"
                    and report["base_commit"] == BASE and report["research_head"] == prepared["research_head"],
                    "native run status/source differs")
    policies = {p["policy"]:p["rows"] for p in report["policies"]}
    recount.require(len(report["policies"]) == 2 and set(policies) == {"immediate","sender-turns-4"},
                    "policy report inventory differs")
    historical = {(r["profile"],r["scenario"]):r for r in recount.read(args.historical_results)}
    metrics,details,age_histograms = {},{},{}
    for policy,minimum in (("immediate",0),("sender-turns-4",MINIMUM)):
        reported = {(r["profile"],r["scenario"]):r for r in policies[policy]}
        recount.require(len(reported) == len(policies[policy]) == 14, "native reported row inventory")
        for profile in recount.PROFILES[:2]:
            for scenario in recount.SCENARIOS:
                path = evidence/policy/(profile+"-"+scenario+".jsonl")
                row,detail = recount.native(path,profile,scenario)
                histogram = verify_ages(recount.rows(path),minimum)
                r = reported[profile,scenario]
                recount.require(sum(r["application"]["committed"]) == row["committed"]
                                and sum(r["application"]["delivered_by_sender"]) == row["delivered"]
                                and r["transport"]["emitted_bytes"] == row["emitted_bytes"]
                                and r["frame_counts"] == detail["frame_counts"]
                                and [len(e["derived_requests"]) for e in r["exposure"]]
                                == [row["derived_owner_A"],row["derived_owner_B"]]
                                and r["trace_sha256"] == detail["trace_sha256"], "reported result differs")
                ages = r["retry_readiness"]
                recount.require(ages["minimum_sender_turns"] == minimum
                                and ages["retry_emissions"] == sum(histogram.values())
                                and ages["minimum_observed_sender_turns"] == (min(histogram) if histogram else None)
                                and ages["maximum_observed_sender_turns"] == (max(histogram) if histogram else None),
                                "retry age report differs")
                key = policy,profile,scenario
                metrics[key],details[key],age_histograms[key] = row,detail,histogram
    comparisons = []
    for profile in recount.PROFILES[:2]:
        for scenario in recount.SCENARIOS:
            baseline = metrics["immediate",profile,scenario]
            aged = metrics["sender-turns-4",profile,scenario]
            differences = {key:{"historical":historical[profile,scenario][key],"fresh":value}
                           for key,value in baseline.items() if historical[profile,scenario][key] != value}
            comparisons.append({"profile":profile,"scenario":scenario,"baseline":baseline,
                                "sender_turns_4":aged,"fresh_baseline_differences_from_retained":differences,
                                "delta":{key:aged[key]-baseline[key] for key in (
                                    "committed","delivered","uncommitted","post_capture_committed",
                                    "derived_owner_A","derived_owner_B","emitted_bytes",
                                    "ack_frames","control_frames","retry_application_frames",
                                    "all_exact_frame_retransmissions","durable_image_revision_increments",
                                    "physical_sync_data")}})
    result = {"schema":1,"status":"passed","analysis":"independent public native-trace recount",
              "base_commit":BASE,"research_head":prepared["research_head"],"traces":28,
              "public_evidence_files_verified":29,"drivers":drivers,"executable":executable,
              "script_sha256":recount.digest(Path(__file__)),
              "recount_source_sha256":recount.digest(Path(recount.__file__)),
              "historical_results_sha256":recount.digest(args.historical_results),
              "comparisons":comparisons,
              "details":[{"policy":key[0],"retry_age_histogram":age_histograms[key],**details[key]}
                         for key in details],
              "interpretation":"All quantities use actual fresh execution; the policy changes public retry readiness. Service, exposure, bytes and durability are reported per schedule; no blanket dominance is assumed."}
    args.output.parent.mkdir(parents=True,exist_ok=True)
    args.output.write_text(json.dumps(result,indent=2,sort_keys=True)+"\n")
    print(json.dumps({"status":"passed","traces":28,
                      "fresh_baseline_rows_equal_to_retained":sum(not c["fresh_baseline_differences_from_retained"] for c in comparisons)},sort_keys=True))


if __name__=="__main__":
    main()
